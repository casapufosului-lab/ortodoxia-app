/**
 * Playback decision logic and YouTube URL construction.
 *
 * Pure functions that determine HOW a source should be opened based on
 * the backend status response and the source's permanent configuration.
 *
 * Three playback modes:
 *   EMBED            → In-app WebView with official YouTube embed (verified + embeddable)
 *   EXTERNAL_CURRENT → Open the verified current live video on YouTube externally
 *   OFFICIAL_SOURCE  → Open the stable official source page (default fallback)
 *
 * These are intentionally separated from UI code so they can be tested
 * without React rendering.
 */

import type { SourceStatusResponse } from './live-status-client';
import type { PermanentLiveSource } from './types';

// ─── Playback mode ─────────────────────────────────────────────────────────

export type PlaybackMode = 'EMBED' | 'EXTERNAL_CURRENT' | 'OFFICIAL_SOURCE';

/**
 * Determine the playback mode for a source given its current backend status.
 *
 * Rules:
 *   - LIVE + embeddable → EMBED (official YouTube iframe)
 *   - LIVE + not embeddable → EXTERNAL_CURRENT (open YouTube externally)
 *   - LIVE + no broadcast info → OFFICIAL_SOURCE (should not happen, safe default)
 *   - NOT_LIVE → OFFICIAL_SOURCE
 *   - UNKNOWN → OFFICIAL_SOURCE
 *   - any error/no data → OFFICIAL_SOURCE
 */
export function determinePlaybackMode(
  sourceStatus: SourceStatusResponse | null,
  _source: PermanentLiveSource,
): PlaybackMode {
  if (!sourceStatus) return 'OFFICIAL_SOURCE';

  const { status, currentBroadcast } = sourceStatus;

  if (status === 'live' && currentBroadcast?.embeddable) {
    return 'EMBED';
  }

  if (status === 'live' && currentBroadcast && !currentBroadcast.embeddable) {
    return 'EXTERNAL_CURRENT';
  }

  return 'OFFICIAL_SOURCE';
}

// ─── YouTube URL construction ──────────────────────────────────────────────

/**
 * Build an official YouTube embed URL for a verified live videoId.
 *
 * Uses only the official youtube.com/embed/ endpoint.
 * autoplay=0 ensures no autoplay on screen open.
 * rel=0 limits related videos to the same channel.
 */
export function buildYouTubeEmbedUrl(videoId: string): string {
  return `https://www.youtube.com/embed/${encodeURIComponent(videoId)}?autoplay=0&rel=0`;
}

/**
 * Build an official YouTube watch URL for opening externally.
 * Used when the video is not embeddable.
 */
export function buildYouTubeWatchUrl(videoId: string): string {
  return `https://www.youtube.com/watch?v=${encodeURIComponent(videoId)}`;
}

import { openBrowserAsync, WebBrowserPresentationStyle } from 'expo-web-browser';

import { isAllowedDomain } from './live-services';
import type { PermanentLiveSource } from './types';

/**
 * Safe opening of official source URLs for "Slujbe LIVE".
 *
 * Only HTTPS URLs on the allowlist may be opened. Malformed or unapproved
 * URLs fail safely and never crash the app. Ortodoxia is NOT the broadcaster
 * and never downloads/proxies/restreams video — it only opens the official
 * provider's own page.
 */

/** User-facing error code when a source cannot be opened. */
export const LIVE_OPEN_ERROR_CODE = 'OPEN_FAILED';

/** Validate that a URL is HTTPS and on the allowlist. */
export function isValidSourceUrl(rawUrl: string): boolean {
  if (typeof rawUrl !== 'string' || rawUrl.length === 0) return false;

  let parsed: URL;
  try {
    parsed = new URL(rawUrl.trim());
  } catch {
    return false;
  }

  if (parsed.protocol !== 'https:') return false;
  if (!isAllowedDomain(parsed.hostname)) return false;

  return true;
}

/**
 * Attempt to open a permanent source's official channel/page.
 * Returns a human-readable error code on failure, or null on success.
 */
export async function openLiveSourceUrl(rawUrl: string): Promise<string | null> {
  if (!isValidSourceUrl(rawUrl)) return LIVE_OPEN_ERROR_CODE;

  try {
    await openBrowserAsync(rawUrl.trim(), { presentationStyle: WebBrowserPresentationStyle.AUTOMATIC });
    return null;
  } catch {
    return LIVE_OPEN_ERROR_CODE;
  }
}

/** Convenience: pick which URL to open for a source (channel preferred, else website). */
export function getOpenableUrlForSource(source: PermanentLiveSource): string | null {
  const candidates = [source.officialChannelUrl, source.officialWebsiteUrl];
  for (const url of candidates) {
    if (isValidSourceUrl(url)) return url;
  }
  return null;
}

/**
 * Client-side live-status fetching and validation.
 *
 * Calls the `live-services-status` Edge Function via Supabase Functions
 * and validates the response shape before rendering. Never exposes any
 * YouTube API key — all discovery happens server-side.
 *
 * This is the ONLY network call the client makes for live detection.
 * A single efficient request fetches all enabled sources at once.
 */

import { supabase } from '@/lib/supabase';

import type { LiveStatus } from './types';

// ─── Response types ────────────────────────────────────────────────────────

/** A single source's live status as returned by the sanitized backend. */
export interface SourceStatusResponse {
  sourceId: string;
  status: LiveStatus;
  checkedAt: string;
  currentBroadcast: {
    videoId: string;
    title: string;
    embeddable: boolean;
    liveStartedAt?: string;
  } | null;
}

/** Shape of the full Edge Function response. */
export interface LiveStatusResult {
  sources: SourceStatusResponse[];
}

// ─── Validation ────────────────────────────────────────────────────────────

const VALID_STATUSES = new Set<string>(['live', 'not_live', 'unknown']);

function isLiveStatus(value: unknown): value is LiveStatus {
  return typeof value === 'string' && VALID_STATUSES.has(value);
}

/**
 * Validate a single source status object from the backend response.
 * Returns null for malformed data instead of throwing.
 */
export function validateSourceStatus(data: unknown): SourceStatusResponse | null {
  if (!data || typeof data !== 'object') return null;

  const obj = data as Record<string, unknown>;

  if (typeof obj.sourceId !== 'string') return null;
  if (!isLiveStatus(obj.status)) return null;
  if (typeof obj.checkedAt !== 'string') return null;

  // currentBroadcast may be null (no live broadcast known)
  if (obj.currentBroadcast != null) {
    if (typeof obj.currentBroadcast !== 'object') return null;
    const cb = obj.currentBroadcast as Record<string, unknown>;
    if (typeof cb.videoId !== 'string') return null;
    if (typeof cb.title !== 'string') return null;
    if (typeof cb.embeddable !== 'boolean') return null;
  }

  return {
    sourceId: obj.sourceId,
    status: obj.status as LiveStatus,
    checkedAt: obj.checkedAt,
    currentBroadcast: obj.currentBroadcast
      ? {
          videoId: (obj.currentBroadcast as Record<string, unknown>).videoId as string,
          title: (obj.currentBroadcast as Record<string, unknown>).title as string,
          embeddable: (obj.currentBroadcast as Record<string, unknown>).embeddable as boolean,
          liveStartedAt: (obj.currentBroadcast as Record<string, unknown>).liveStartedAt as
            | string
            | undefined,
        }
      : null,
  };
}

// ─── Fetching ──────────────────────────────────────────────────────────────

/** User-facing error code for backend failure. */
export const LIVE_STATUS_FETCH_ERROR_CODE = 'FETCH_FAILED';

/**
 * Fetch live statuses for all enabled sources in a single Edge Function call.
 *
 * - Never calls YouTube Data API directly.
 * - Never exposes API keys in client code.
 * - Malformed data fails safely (filtered out, never crashes).
 * - No duplicate requests from rerenders (caller manages invocation).
 */
export async function fetchLiveStatuses(countryCode: string = 'RO'): Promise<LiveStatusResult> {
  const { data, error } = await supabase.functions.invoke('live-services-status', {
    method: 'GET',
    headers: { 'x-country-code': countryCode },
  });

  if (error || !data) {
    console.warn('live-services-status fetch failed:', error?.message ?? 'no data');
    throw new Error(LIVE_STATUS_FETCH_ERROR_CODE);
  }

  if (!data.sources || !Array.isArray(data.sources)) {
    console.warn('live-services-status: unexpected response shape');
    throw new Error(LIVE_STATUS_FETCH_ERROR_CODE);
  }

  const sources = (data.sources as unknown[])
    .map(validateSourceStatus)
    .filter((s): s is SourceStatusResponse => s !== null);

  return { sources };
}

/**
 * Core type definitions for the "Slujbe LIVE" feature.
 *
 * These types separate the PERMANENT SOURCE (the stable identity of a
 * church/monastery/provider/broadcaster) from the CURRENT LIVE BROADCAST
 * (a transient live video event). LIVE-B and LIVE-C will plug dynamic
 * detection into this architecture WITHOUT rewriting the UI model.
 */

/** Playback destination supported by a permanent source. */
export type StreamType = 'youtube' | 'official_link';

/**
 * Live status supported by the UI model.
 * This task renders UNKNOWN/neutral; LIVE-B/LIVE-C will connect real detection.
 */
export type LiveStatus = 'live' | 'not_live' | 'unknown';

/**
 * PERMANENT SOURCE — stable identity of a broadcaster/provider.
 *
 * This object describes WHO broadcasts liturgies, their stable official
 * website and stable official YouTube channel. It must NEVER contain a
 * one-off YouTube `watch?v=...` videoId as permanent configuration.
 */
export interface PermanentLiveSource {
  /** Stable unique identifier for this source (e.g. 'trinitas-tv'). */
  id: string;

  /** App country code that owns this source (RO, MD, GR, etc.). */
  countryCode: string;

  /** Public-facing source/church/monastery name. */
  name: string;

  /** City/region where the broadcaster is based. */
  location: string;

  /** Organization that operates the source (e.g. 'Patriarhia Română'). */
  provider: string;

  /** Stable official website (HTTPS). */
  officialWebsiteUrl: string;

  /** Stable official YouTube channel page (HTTPS). */
  officialChannelUrl: string;

  /** Verified official YouTube channel ID (UC...) — only when confidently verified. */
  officialChannelId?: string;

  /** How this source is watched. */
  streamType: StreamType;

  /** Sort priority (lower = shown first among enabled sources). */
  priority: number;

  /** Optional verified schedule note (only when supported by official evidence). */
  scheduleNote?: string;

  /** Whether this source is currently wired in. Disabled sources are excluded from UI. */
  enabled: boolean;

  /** Whether LIVE-B can attempt dynamic live detection against this channel. */
  supportsDynamicLiveDetection: boolean;
}

/**
 * CURRENT LIVE BROADCAST — transient data about a live event.
 *
 * Created only later (LIVE-B) from YouTube API discovery. The app must
 * NEVER derive a permanent source from one of these transient objects.
 */
export interface CurrentLiveBroadcast {
  /** The live video's temporary YouTube videoId. */
  videoId: string;

  /** Derived live status. */
  status: LiveStatus;

  /** ISO timestamp of when this status was last checked. */
  checkedAt: string;

  /** Whether the widget/browser can embed this broadcast. */
  embeddable: boolean;

  /** Display title of the live event. */
  title: string;

  /** Optional extra metadata from the platform. */
  metadata?: Record<string, unknown>;
}

/**
 * YouTube player for native platforms.
 *
 * Uses the official YouTube IFrame Player API through react-native-youtube-iframe.
 * Normal playback stays in portrait.
 * Android fullscreen rotates to landscape only after the user requests fullscreen.
 */

import { useCallback, useEffect } from 'react';
import * as ScreenOrientation from 'expo-screen-orientation';
import { BackHandler, Modal, StyleSheet, useWindowDimensions, View } from 'react-native';
import YoutubePlayer from 'react-native-youtube-iframe';

import { useLocalization } from '@/providers/localization-provider';


export interface YouTubeEmbedProps {
  videoId: string | null;
  onClose: () => void;
}

export function YouTubeEmbed({ videoId, onClose }: YouTubeEmbedProps) {
  const { t } = useLocalization();
  const { width } = useWindowDimensions();

  const handleBack = useCallback(() => {
    if (videoId) {
      onClose();
      return true;
    }
    return false;
  }, [videoId, onClose]);

  useEffect(() => {
    const sub = BackHandler.addEventListener('hardwareBackPress', handleBack);
    return () => sub.remove();
  }, [handleBack]);

  const handleFullScreenChange = useCallback((isFullScreen: boolean) => {
    const orientation = isFullScreen
      ? ScreenOrientation.OrientationLock.LANDSCAPE
      : ScreenOrientation.OrientationLock.PORTRAIT_UP;

    void ScreenOrientation.lockAsync(orientation).catch(() => undefined);
  }, []);

  useEffect(() => {
    if (!videoId) return;

    return () => {
      void ScreenOrientation.lockAsync(
        ScreenOrientation.OrientationLock.PORTRAIT_UP,
      ).catch(() => undefined);
    };
  }, [videoId]);

  if (!videoId) return null;

  const playerHeight = Math.max(220, Math.round(width * 9 / 16));

  return (
    <Modal
      visible
      animationType="slide"
      presentationStyle="pageSheet"
      onRequestClose={onClose}
    >
      <View style={styles.container}>
        <View style={styles.playerArea}>
          <YoutubePlayer
            height={playerHeight}
            videoId={videoId}
            onFullScreenChange={handleFullScreenChange}
            initialPlayerParams={{
              controls: true,
              preventFullScreen: false,
              rel: false,
            }}
            webViewProps={{
              allowsFullscreenVideo: true,

              accessibilityLabel: t('live.playerAccessibilityLabel'),
            }}
          />
        </View>
      </View>
    </Modal>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#000' },
  playerArea: {
    flex: 1,
    justifyContent: 'center',
    backgroundColor: '#000',
  },
});

function createAdminClient(url: string, key: string) {
  return createClient(url, key, {
    auth: { persistSession: false, autoRefreshToken: false },
  });
}

function createSupacheCacheAdapter(
  admin: ReturnType<typeof createAdminClient>,
): CacheStore {
  return {
    async get(sourceId: string): Promise<CachedStatusRow | null> {
      const { data, error } = await admin
        .from('live_service_status_cache')
        .select('*')
        .eq('source_id', sourceId)
        .maybeSingle();
      if (error || !data) return null;
      return data as CachedStatusRow;
    },

    async upsert(row: Partial<CachedStatusRow> & { source_id: string }): Promise<void> {
      const { error } = await admin
        .from('live_service_status_cache')
        .upsert(
          { ...row, updated_at: new Date().toISOString() },
          { onConflict: 'source_id' },
        );
      if (error) {
        console.error('live-cache upsert failed', row.source_id, error.message);
      }
    },

    async tryClaimLease(sourceId: string, nowMs: number): Promise<boolean> {
      const leaseUntil = new Date(nowMs + 60_000).toISOString();
      const now = new Date(nowMs).toISOString();

      // Seed a baseline row (ON CONFLICT DO NOTHING) so the conditional UPDATE
      // below can match even on the very first request for a source. Without
      // this, the lease could never be claimed for a brand-new source and the
      // discovery would be skipped forever (always UNKNOWN).
      const { error: seedError } = await admin
        .from('live_service_status_cache')
        .upsert(
          {
            source_id: sourceId,
            status: 'unknown',
            current_video_id: null,
            current_title: null,
            current_channel_id: null,
            embeddable: null,
            checked_at: now,
            live_started_at: null,
            next_refresh_at: now,
            refresh_lease_until: null,
            updated_at: now,
          },
          { onConflict: 'source_id', ignoreDuplicates: true },
        );
      if (seedError) {
        console.error('live-cache lease seed failed', sourceId, seedError.message);
        return false;
      }

      // Atomic: only update if no active lease exists
      const { data, error } = await admin
        .from('live_service_status_cache')
        .update({ refresh_lease_until: leaseUntil, updated_at: now })
        .or(`refresh_lease_until.is.null,refresh_lease_until.lt.${now}`)
        .eq('source_id', sourceId)
        .select('source_id')
        .maybeSingle();

      if (error) {
        console.error('live-cache lease failed', sourceId, error.message);
        return false;
      }
      return Boolean(data);
    },
  };
}

/**
 * Live status engine — orchestration for dynamic YouTube live detection.
 *
 * Contains all timing constants and the status computation logic.
 * Dependency-injected for testability: callers supply YouTube client + cache store.
 *
 * Invariants:
 *   - UNKNOWN never silently becomes NOT_LIVE.
 *   - One-off videoId never enters permanent source configuration.
 *   - Disabled sources cause zero YouTube discovery quota usage.
 *   - Concurrent stale callers reuse cache or UNKNOWN (no duplicate searches).
 */

import type { YouTubeClient, LiveSearchResult, VideoVerification } from './youtube.ts';

// ─── Timing constants (centralised) ────────────────────────────────────────

export const DISCOVERY_REFRESH_MS = 90 * 60 * 1000;
export const LIVE_REVALIDATE_MS = 5 * 60 * 1000;
export const LEASE_MS = 60 * 1000;
export const STALE_LIVE_REUSE_MS = 5 * 60 * 1000;

// ─── Status model ───────────────────────────────────────────────────────────

export type LiveStatus = 'live' | 'not_live' | 'unknown';

export type CachedStatusRow = {
  source_id: string;
  status: LiveStatus;
  current_video_id: string | null;
  current_title: string | null;
  current_channel_id: string | null;
  embeddable: boolean | null;
  checked_at: string;
  live_started_at: string | null;
  next_refresh_at: string;
  refresh_lease_until: string | null;
};

export type SanitizedResponse = {
  sourceId: string;
  status: LiveStatus;
  checkedAt: string;
  currentBroadcast: {
    videoId: string;
    title: string;
    embeddable: boolean;
    liveStartedAt: string | undefined;
  } | null;
};

export type SourceDef = {

  officialChannelId: string;
  enabled: boolean;
  supportsDynamicLiveDetection: boolean;
};

export interface CacheStore {
  get(sourceId: string): Promise<CachedStatusRow | null>;
  upsert(row: Partial<CachedStatusRow> & { source_id: string }): Promise<void>;
  tryClaimLease(sourceId: string, nowMs: number): Promise<boolean>;
}

// ─── YouTube discovery ──────────────────────────────────────────────────────

export type YouTubeDiscoveryOutcome = {
  status: 'LIVE' | 'NOT_LIVE' | 'UNKNOWN';
  videoId?: string;
  channelId?: string;
  title?: string;
  embeddable?: boolean;
  liveStartedAt?: string;
};

/**
 * Discover whether a source is currently live.
 * Steps: search.list → videos.list verify → reject wrong channel / ended.
 * Any error → UNKNOWN (never silently NOT_LIVE).
 */
export async function discoverLiveStatus(
  youtube: YouTubeClient,
  source: SourceDef,
): Promise<YouTubeDiscoveryOutcome> {
  let candidates: LiveSearchResult[];
  try {
    candidates = await youtube.searchLive(source.officialChannelId);
  } catch {
    return { status: 'UNKNOWN' };
  }

  if (candidates.length === 0) {
    return { status: 'NOT_LIVE' };
  }

  let sawUnverifiedCandidate = false;

  for (const candidate of candidates) {
    if (candidate.channelId !== source.officialChannelId) continue;

    let verification: VideoVerification | null;
    try {
      verification = await youtube.verifyVideo(candidate.videoId);
    } catch {
      return { status: 'UNKNOWN' };
    }

    if (!verification) continue;
    if (verification.channelId !== source.officialChannelId) continue;
    if (verification.actualEndTime) continue;
    if (!verification.isLiveContent) continue;
    if (!verification.actualStartTime) {
      sawUnverifiedCandidate = true;
      continue;
    }

    return {
      status: 'LIVE',
      videoId: verification.videoId,
      channelId: verification.channelId,
      title: verification.title,
      embeddable: verification.embeddable,
      liveStartedAt: verification.actualStartTime,
    };
  }

  return { status: sawUnverifiedCandidate ? 'UNKNOWN' : 'NOT_LIVE' };
}

// ─── Sanitise cache → response ─────────────────────────────────────────────

export function sanitise(row: CachedStatusRow | null, sourceId: string): SanitizedResponse {
  if (!row) {
    return { sourceId, status: 'unknown', checkedAt: '', currentBroadcast: null };
  }
  return {
    sourceId: row.source_id,
    status: row.status,
    checkedAt: row.checked_at,
    currentBroadcast:
      row.status === 'live' && row.current_video_id
        ? {
            videoId: row.current_video_id,
            title: row.current_title ?? '',
            embeddable: row.embeddable ?? false,
            liveStartedAt: row.live_started_at ?? undefined,
          }
        : null,
  };
}

// ─── Orchestration ──────────────────────────────────────────────────────────

function unknownResponse(sourceId: string, nowMs: number): SanitizedResponse {
  return {
    sourceId,
    status: 'unknown',
    checkedAt: new Date(nowMs).toISOString(),
    currentBroadcast: null,
  };
}

/**
 * Compute status for a single enabled, eligible source.
 * Uses cache + single-flight lease to prevent thundering-herd.
 */
export async function computeSourceStatus(
  source: SourceDef,
  youtube: YouTubeClient | null,
  cache: CacheStore,
  nowMs: number,
  apiKeyPresent: boolean,
): Promise<SanitizedResponse> {
  const row = await cache.get(source.id);

  // Cache is fresh → reuse directly
  if (row && new Date(row.next_refresh_at).getTime() > nowMs) {
    return sanitise(row, source.id);
  }

  // No API key → everything UNKNOWN
  if (!apiKeyPresent || !youtube) {
    await cache.upsert({
      source_id: source.id,
      status: 'unknown',
      current_video_id: null,
      current_title: null,
      current_channel_id: null,
      embeddable: null,
      checked_at: new Date(nowMs).toISOString(),
      live_started_at: null,
      next_refresh_at: new Date(nowMs + DISCOVERY_REFRESH_MS).toISOString(),
      refresh_lease_until: null,
    });
    return unknownResponse(source.id, nowMs);
  }

  // Stale: try single-flight lease
  const leaseClaimed = await cache.tryClaimLease(source.id, nowMs);
  if (!leaseClaimed) {
    if (row?.status === 'live' && row.current_video_id) {
      const ageMs = nowMs - new Date(row.checked_at).getTime();
      if (ageMs <= STALE_LIVE_REUSE_MS) {
        return sanitise(row, source.id);
      }
    }
    return unknownResponse(source.id, nowMs);
  }

  // We hold the lease → YouTube discovery
  const outcome = await discoverLiveStatus(youtube, source);

  if (outcome.status === 'LIVE' && outcome.videoId) {
    await cache.upsert({
      source_id: source.id,
      status: 'live',
      current_video_id: outcome.videoId,
      current_title: outcome.title ?? null,
      current_channel_id: outcome.channelId ?? null,
      embeddable: outcome.embeddable ?? null,
      checked_at: new Date(nowMs).toISOString(),
      live_started_at: outcome.liveStartedAt ?? null,
      next_refresh_at: new Date(nowMs + LIVE_REVALIDATE_MS).toISOString(),
      refresh_lease_until: null,
    });
    return sanitise(
      {
        source_id: source.id,
        status: 'live',
        current_video_id: outcome.videoId,
        current_title: outcome.title ?? null,
        current_channel_id: outcome.channelId ?? null,
        embeddable: outcome.embeddable ?? null,
        checked_at: new Date(nowMs).toISOString(),
        live_started_at: outcome.liveStartedAt ?? null,
        next_refresh_at: new Date(nowMs + LIVE_REVALIDATE_MS).toISOString(),
        refresh_lease_until: null,
      },
      source.id,
    );
  }

  if (outcome.status === 'NOT_LIVE') {
    const notLiveRow: Partial<CachedStatusRow> & { source_id: string } = {
      source_id: source.id,
      status: 'not_live',
      current_video_id: null,
      current_title: null,
      current_channel_id: null,
      embeddable: null,
      checked_at: new Date(nowMs).toISOString(),
      live_started_at: null,
      next_refresh_at: new Date(nowMs + DISCOVERY_REFRESH_MS).toISOString(),
      refresh_lease_until: null,
    };
    await cache.upsert(notLiveRow);
    return sanitise(
      { ...notLiveRow, refresh_lease_until: null } as CachedStatusRow,
      source.id,
    );
  }

  // UNKNOWN: if we had a recent LIVE cache, preserve it
  if (row?.status === 'live' && row.current_video_id) {
    const ageMs = nowMs - new Date(row.checked_at).getTime();
    if (ageMs <= STALE_LIVE_REUSE_MS) {
      await cache.upsert({
        source_id: source.id,
        status: 'live',
        current_video_id: row.current_video_id,
        current_title: row.current_title,
        current_channel_id: row.current_channel_id,
        embeddable: row.embeddable,
        checked_at: row.checked_at,
        live_started_at: row.live_started_at,
        next_refresh_at: new Date(nowMs + LIVE_REVALIDATE_MS).toISOString(),
        refresh_lease_until: null,
      });
      return sanitise(row, source.id);
    }
  }

  await cache.upsert({
    source_id: source.id,
    status: 'unknown',
    current_video_id: null,
    current_title: null,
    current_channel_id: null,
    embeddable: null,
    checked_at: new Date(nowMs).toISOString(),
    live_started_at: null,
    next_refresh_at: new Date(nowMs + DISCOVERY_REFRESH_MS).toISOString(),
    refresh_lease_until: null,
  });
  return unknownResponse(source.id, nowMs);
}

/**
 * Compute statuses for all eligible sources.
 */
export async function computeAllSourceStatuses(
  sources: readonly SourceDef[],
  youtube: YouTubeClient | null,
  cache: CacheStore,
  nowMs: number,
  apiKeyPresent: boolean,
): Promise<SanitizedResponse[]> {
  const results: SanitizedResponse[] = [];
  for (const source of sources) {
    results.push(
      await computeSourceStatus(source, youtube, cache, nowMs, apiKeyPresent),
    );
  }
  return results;
}

/**
 * YouTube Data API v3 client — the ONLY component that talks to YouTube.
 *
 * Uses ONLY official documented endpoints:
 *   - search.list  (discovery: find live videos on a channel)
 *   - videos.list  (revalidation: verify candidate video status)
 *
 * All requests require a valid YOUTUBE_DATA_API_KEY.
 * Never scrapes HTML, never uses internal/undocumented endpoints.
 */

export type LiveSearchResult = {
  videoId: string;
  channelId: string;
  title: string;
};

export type VideoVerification = {
  videoId: string;
  channelId: string;
  title: string;
  embeddable: boolean;
  isLiveContent: boolean;
  actualStartTime: string | undefined;
  actualEndTime: string | undefined;
};

export type YouTubeClient = {
  searchLive(channelId: string): Promise<LiveSearchResult[]>;
  verifyVideo(videoId: string): Promise<VideoVerification | null>;
};

const YOUTUBE_API_BASE = 'https://www.googleapis.com/youtube/v3';

type SearchListResponse = {
  items?: Array<{
    id?: { videoId?: string };
    snippet?: { channelId?: string; title?: string };
  }>;
  error?: { message?: string; code?: number };
};

type VideoItem = {
  id?: string;
  snippet?: { channelId?: string; title?: string };
  liveStreamingDetails?: { actualStartTime?: string; actualEndTime?: string };
  status?: { embeddable?: boolean };
  contentDetails?: { relatedPlaylists?: Record<string, string> };
};

type VideoListResponse = {
  items?: VideoItem[];
  error?: { message?: string; code?: number };
};

/**
 * Create a YouTube client bound to a specific API key.
 *
 * @param apiKey  The YouTube Data API key (server-side secret).
 * @param fetchFn Injectable fetch for testability (defaults to global fetch).
 */
export function createYouTubeClient(
  apiKey: string,
  fetchFn: typeof fetch = globalThis.fetch,
): YouTubeClient {
  async function request<T>(path: string, params: Record<string, string>): Promise<T> {
    const url = new URL(`${YOUTUBE_API_BASE}${path}`);
    url.searchParams.set('key', apiKey);
    for (const [k, v] of Object.entries(params)) {
      url.searchParams.set(k, v);
    }

    const res = await fetchFn(url.toString(), {
      method: 'GET',
      signal: AbortSignal.timeout(10_000),
    });
    if (!res.ok) {
      const body = await res.json().catch(() => null) as Record<string, unknown> | null;
      const errMsg = (body as { error?: { message?: string } })?.error?.message ?? `HTTP ${res.status}`;
      throw new Error(`YouTube API error: ${errMsg}`);
    }
    return (await res.json()) as T;
  }

  return {
    async searchLive(channelId: string): Promise<LiveSearchResult[]> {

      const data = await request<SearchListResponse>('/search', {
        part: 'snippet',
        channelId,
        eventType: 'live',
        type: 'video',
        maxResults: '5',
        order: 'date',
      });

      if (data.error) {
        throw new Error(`YouTube search error: ${data.error.message ?? 'unknown'}`);
      }

      if (!Array.isArray(data.items)) {
        throw new Error('YouTube search response malformed');
      }

      return data.items
        .map((item) => ({
          videoId: item.id?.videoId ?? '',
          channelId: item.snippet?.channelId ?? '',
          title: item.snippet?.title ?? '',
        }))
        .filter((r) => r.videoId && r.channelId);
    },

    async verifyVideo(videoId: string): Promise<VideoVerification | null> {
      const data = await request<VideoListResponse>('/videos', {
        part: 'snippet,liveStreamingDetails,status',
        id: videoId,
        maxResults: '1',
      });

      if (data.error) {
        throw new Error(`YouTube videos error: ${data.error.message ?? 'unknown'}`);
      }

      if (!Array.isArray(data.items)) {
        throw new Error('YouTube videos response malformed');
      }

      const item = data.items[0];
      if (!item) return null;
      if (!item.id || !item.snippet?.channelId) {
        throw new Error('YouTube videos response malformed');
      }

      return {
        videoId: item.id,
        channelId: item.snippet?.channelId ?? '',
        title: item.snippet?.title ?? '',
        embeddable: item.status?.embeddable ?? false,
        isLiveContent: Boolean(item.liveStreamingDetails),
        actualStartTime: item.liveStreamingDetails?.actualStartTime,
        actualEndTime: item.liveStreamingDetails?.actualEndTime,
      };
    },
  };
}

function extractClientSources(src) {
  const objects = [];
  const idRe = /\n\s*\{\s*\n\s*id:\s*'([^']+)'/g;
  let m;
  while ((m = idRe.exec(src)) !== null) {
    const start = m.index;
    const end = src.indexOf('\n  },', start);
    if (end === -1) continue;
    objects.push({ id: m[1], block: src.slice(start, end + 4) });
  }
  return objects;
}

function getField(block, field) {
  // String values (quoted) or booleans.
  const str = block.match(new RegExp(`${field}:\\s*'([^']*)'`));
  if (str) return str[1];
  const bool = block.match(new RegExp(`${field}:\\s*(true|false)`));
  if (bool) return bool[1];
  return undefined;
}

// In-memory cache store implementing the CacheStore interface.
function makeMemoryCache(initial = {}) {
  const store = new Map(Object.entries(initial));
  return {
    store,
    async get(sourceId) {
      return store.has(sourceId) ? { ...store.get(sourceId) } : null;
    },
    async upsert(row) {
      store.set(row.source_id, { ...store.get(row.source_id) ?? {}, ...row });
    },
    async tryClaimLease(sourceId, nowMs) {
      // Mirrors the Supabase adapter: the first claim also creates the row
      // (ON CONFLICT DO NOTHING) so brand-new sources can still win the lease.
      const existing = store.get(sourceId);
      const leaseUntil = existing?.refresh_lease_until
        ? new Date(existing.refresh_lease_until).getTime()
        : 0;
      if (leaseUntil > nowMs) return false;
      store.set(sourceId, {
        ...(existing ?? {}),
        source_id: sourceId,
        refresh_lease_until: new Date(nowMs + 60_000).toISOString(),
      });
      return true;
    },
  };
}

function extractAllowlist(src) {
  const match = src.match(/ALLOWED_DOMAINS\s*=\s*\[([\s\S]*?)\]\s*as const/);
  assert.ok(match, 'ALLOWED_DOMAINS array not found');
  const entries = [...match[1].matchAll(/'([^']+)'/g)].map((m) => m[1]);
  return new Set(entries.map((d) => d.toLowerCase()));
}

function extractSourceObjects(src) {
  const objects = [];
  const idRe = /\n\s*\{\s*\n\s*id:\s*'([^']+)'/g;
  let m;
  while ((m = idRe.exec(src)) !== null) {
    const start = m.index;
    const endSearch = src.indexOf('\n  },', start);
    const end = endSearch === -1 ? src.indexOf('\n  }', start) : endSearch;
    if (end === -1) continue;
    objects.push({ id: m[1], block: src.slice(start, end + 4) });
  }
  return objects;
}

function getBlockField(block, field) {
  const lines = block.split('\n');
  for (const line of lines) {

    if (m) return m[1].trim().replace(/,/g, '').replace(/'/g, '');
  }
  return undefined;
}

/**
 * Playback decision logic and YouTube URL construction.
 *
 * Pure functions that determine HOW a source should be opened based on
 * the backend status response and the source's permanent configuration.
 *
 * Three playback modes:
 *   EMBED            → In-app WebView with official YouTube embed (verified + embeddable)
 *   EXTERNAL_CURRENT → Open the verified current live video on YouTube externally
 *   OFFICIAL_SOURCE  → Open the stable official source page (default fallback)
 *
 * These are intentionally separated from UI code so they can be tested
 * without React rendering.
 */

import type { SourceStatusResponse } from './live-status-client';
import type { PermanentLiveSource } from './types';

// ─── Playback mode ─────────────────────────────────────────────────────────

export type PlaybackMode = 'EMBED' | 'EXTERNAL_CURRENT' | 'OFFICIAL_SOURCE';

/**
 * Determine the playback mode for a source given its current backend status.
 *
 * Rules:
 *   - LIVE + embeddable → EMBED (official YouTube iframe)
 *   - LIVE + not embeddable → EXTERNAL_CURRENT (open YouTube externally)
 *   - LIVE + no broadcast info → OFFICIAL_SOURCE (should not happen, safe default)
 *   - NOT_LIVE → OFFICIAL_SOURCE
 *   - UNKNOWN → OFFICIAL_SOURCE
 *   - any error/no data → OFFICIAL_SOURCE
 */
export function determinePlaybackMode(
  sourceStatus: SourceStatusResponse | null,
  _source: PermanentLiveSource,
): PlaybackMode {
  if (!sourceStatus) return 'OFFICIAL_SOURCE';

  const { status, currentBroadcast } = sourceStatus;

  if (status === 'live' && currentBroadcast?.embeddable) {
    return 'EMBED';
  }

  if (status === 'live' && currentBroadcast && !currentBroadcast.embeddable) {
    return 'EXTERNAL_CURRENT';
  }

  return 'OFFICIAL_SOURCE';
}

// ─── YouTube URL construction ──────────────────────────────────────────────

/**
 * Build an official YouTube embed URL for a verified live videoId.
 *
 * Uses only the official youtube.com/embed/ endpoint.
 * autoplay=0 ensures no autoplay on screen open.
 * rel=0 limits related videos to the same channel.
 */
export function buildYouTubeEmbedUrl(videoId: string): string {
  return `https://www.youtube.com/embed/${encodeURIComponent(videoId)}?autoplay=0&rel=0`;
}

/**
 * Build an official YouTube watch URL for opening externally.
 * Used when the video is not embeddable.
 */
export function buildYouTubeWatchUrl(videoId: string): string {
  return `https://www.youtube.com/watch?v=${encodeURIComponent(videoId)}`;
}

import { openBrowserAsync, WebBrowserPresentationStyle } from 'expo-web-browser';

import { isAllowedDomain } from './live-services';
import type { PermanentLiveSource } from './types';

/**
 * Safe opening of official source URLs for "Slujbe LIVE".
 *
 * Only HTTPS URLs on the allowlist may be opened. Malformed or unapproved
 * URLs fail safely and never crash the app. Ortodoxia is NOT the broadcaster
 * and never downloads/proxies/restreams video — it only opens the official
 * provider's own page.
 */

/** User-facing error code when a source cannot be opened. */
export const LIVE_OPEN_ERROR_CODE = 'OPEN_FAILED';

/** Validate that a URL is HTTPS and on the allowlist. */
export function isValidSourceUrl(rawUrl: string): boolean {
  if (typeof rawUrl !== 'string' || rawUrl.length === 0) return false;

  let parsed: URL;
  try {
    parsed = new URL(rawUrl.trim());
  } catch {
    return false;
  }

  if (parsed.protocol !== 'https:') return false;
  if (!isAllowedDomain(parsed.hostname)) return false;

  return true;
}

/**
 * Attempt to open a permanent source's official channel/page.
 * Returns a human-readable error code on failure, or null on success.
 */
export async function openLiveSourceUrl(rawUrl: string): Promise<string | null> {
  if (!isValidSourceUrl(rawUrl)) return LIVE_OPEN_ERROR_CODE;

  try {
    await openBrowserAsync(rawUrl.trim(), { presentationStyle: WebBrowserPresentationStyle.AUTOMATIC });
    return null;
  } catch {
    return LIVE_OPEN_ERROR_CODE;
  }
}

/**
 * Client-side live-status fetching and validation.
 *
 * Calls the `live-services-status` Edge Function via Supabase Functions
 * and validates the response shape before rendering. Never exposes any
 * YouTube API key — all discovery happens server-side.
 *
 * This is the ONLY network call the client makes for live detection.
 * A single efficient request fetches all enabled sources at once.
 */

import { supabase } from '@/lib/supabase';

import type { LiveStatus } from './types';

// ─── Response types ────────────────────────────────────────────────────────

/** A single source's live status as returned by the sanitized backend. */
export interface SourceStatusResponse {
  sourceId: string;
  status: LiveStatus;
  checkedAt: string;
  currentBroadcast: {
    videoId: string;
    title: string;
    embeddable: boolean;
    liveStartedAt?: string;
  } | null;
}

/** Shape of the full Edge Function response. */
export interface LiveStatusResult {
  sources: SourceStatusResponse[];
}

// ─── Validation ────────────────────────────────────────────────────────────

const VALID_STATUSES = new Set<string>(['live', 'not_live', 'unknown']);

function isLiveStatus(value: unknown): value is LiveStatus {
  return typeof value === 'string' && VALID_STATUSES.has(value);
}

/**
 * Validate a single source status object from the backend response.
 * Returns null for malformed data instead of throwing.
 */
export function validateSourceStatus(data: unknown): SourceStatusResponse | null {
  if (!data || typeof data !== 'object') return null;

  const obj = data as Record<string, unknown>;

  if (typeof obj.sourceId !== 'string') return null;
  if (!isLiveStatus(obj.status)) return null;
  if (typeof obj.checkedAt !== 'string') return null;

  // currentBroadcast may be null (no live broadcast known)
  if (obj.currentBroadcast != null) {
    if (typeof obj.currentBroadcast !== 'object') return null;
    const cb = obj.currentBroadcast as Record<string, unknown>;
    if (typeof cb.videoId !== 'string') return null;
    if (typeof cb.title !== 'string') return null;
    if (typeof cb.embeddable !== 'boolean') return null;
  }

  return {
    sourceId: obj.sourceId,
    status: obj.status as LiveStatus,
    checkedAt: obj.checkedAt,
    currentBroadcast: obj.currentBroadcast
      ? {
          videoId: (obj.currentBroadcast as Record<string, unknown>).videoId as string,
          title: (obj.currentBroadcast as Record<string, unknown>).title as string,
          embeddable: (obj.currentBroadcast as Record<string, unknown>).embeddable as boolean,
          liveStartedAt: (obj.currentBroadcast as Record<string, unknown>).liveStartedAt as
            | string
            | undefined,
        }
      : null,
  };
}

// ─── Fetching ──────────────────────────────────────────────────────────────

/** User-facing error code for backend failure. */
export const LIVE_STATUS_FETCH_ERROR_CODE = 'FETCH_FAILED';

/**
 * Fetch live statuses for all enabled sources in a single Edge Function call.
 *
 * - Never calls YouTube Data API directly.
 * - Never exposes API keys in client code.
 * - Malformed data fails safely (filtered out, never crashes).
 * - No duplicate requests from rerenders (caller manages invocation).
 */
export async function fetchLiveStatuses(countryCode: string = 'RO'): Promise<LiveStatusResult> {
  const { data, error } = await supabase.functions.invoke('live-services-status', {
    method: 'GET',
    headers: { 'x-country-code': countryCode },
  });

  if (error || !data) {
    console.warn('live-services-status fetch failed:', error?.message ?? 'no data');
    throw new Error(LIVE_STATUS_FETCH_ERROR_CODE);
  }

  if (!data.sources || !Array.isArray(data.sources)) {
    console.warn('live-services-status: unexpected response shape');
    throw new Error(LIVE_STATUS_FETCH_ERROR_CODE);
  }

  const sources = (data.sources as unknown[])
    .map(validateSourceStatus)
    .filter((s): s is SourceStatusResponse => s !== null);

  return { sources };
}

/**

 *
 * These types separate the PERMANENT SOURCE (the stable identity of a
 * church/monastery/provider/broadcaster) from the CURRENT LIVE BROADCAST
 * (a transient live video event). LIVE-B and LIVE-C will plug dynamic

 */

export type StreamType = 'youtube' | 'official_link';

/**
 * Live status supported by the UI model.
 * This task renders UNKNOWN/neutral; LIVE-B/LIVE-C will connect real detection.
 */
export type LiveStatus = 'live' | 'not_live' | 'unknown';

/**
 * PERMANENT SOURCE — stable identity of a broadcaster/provider.
 *
 * This object describes WHO broadcasts liturgies, their stable official
 * website and stable official YouTube channel. It must NEVER contain a
 * one-off YouTube `watch?v=...` videoId as permanent configuration.
 */
export interface PermanentLiveSource {

  id: string;

  countryCode: string;

  name: string;

  location: string;

  /** Organization that operates the source (e.g. 'Patriarhia Română'). */
  provider: string;

  /** Stable official website (HTTPS). */
  officialWebsiteUrl: string;

  /** Stable official YouTube channel page (HTTPS). */
  officialChannelUrl: string;

  officialChannelId?: string;

  streamType: StreamType;

  /** Sort priority (lower = shown first among enabled sources). */
  priority: number;

  /** Optional verified schedule note (only when supported by official evidence). */
  scheduleNote?: string;

  enabled: boolean;

  /** Whether LIVE-B can attempt dynamic live detection against this channel. */
  supportsDynamicLiveDetection: boolean;
}

  videoId: string;

  status: LiveStatus;

  checkedAt: string;

  /** Whether the widget/browser can embed this broadcast. */
  embeddable: boolean;

  /** Display title of the live event. */
  title: string;

  /** Optional extra metadata from the platform. */
  metadata?: Record<string, unknown>;
}
